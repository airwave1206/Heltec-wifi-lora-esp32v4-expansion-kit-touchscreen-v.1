handheld weather station rtc upgrade
---------------------------------------

o add the DS3231 RTC Module to your system, we need to install the RTClib library by Adafruit via the Arduino Library Manager.
The code below implements a hybrid timekeeping system:
1. On boot, the clock instantly displays the correct time pulled from the RTC chip (no more starting at 8:00 AM).
2. When you take the device outside and get a valid GPS lock, the code automatically syncs the high-accuracy satellite time directly into the RTC chip to correct any long-term drift.
Here are the code updates for your project:


1. Add the Library and Object Definition
----------------------------------------------
Add this to the very top of your main sketch file, where your other #include statements are located:

cpp
#include <Wire.h>
#include "RTClib.h"

RTC_DS3231 rtc;
bool rtcFound = false;


2. Initialize the RTC in setup()
------------------------------------------

Scroll down to your setup() function. Find where your I2C buses are initialized (Wire.begin), and paste the RTC startup code directly underneath it:
cpp
    // ... [Inside setup() after your Wire.begin initialization] ...
    Serial.println("Initializing I2C Wire (Sensors)...");
    Wire.begin(B2B_SDA, B2B_SCL, 100000);         
    
    // --- NEW: INITIALIZE RTC MODULE ---
    Serial.println("Initializing DS3231 RTC...");
    if (!rtc.begin(&Wire)) {
        Serial.println("[ERROR] Couldn't find DS3231 RTC chip!");
        rtcFound = false;
    } else {
        Serial.println("[OK] DS3231 RTC communication online.");
        rtcFound = true;
        
        // Optional safety: If the RTC lost power (battery removed), seed it with compile time
        if (rtc.lostPower()) {
            Serial.println("[WARN] RTC lost power! Seeding with compile time...");
            rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
        }
    }
    // ----------------------------------

    Serial.println("Initializing I2C Wire1 (Touch)...");
    Wire1.begin(TOUCH_SDA_PIN, TOUCH_SCL_PIN); 

  
3. Replace the Time-Keeping Block in loop()
---------------------------------------------------
Inside your loop() function, find the entire // --- SYSTEM TIME CALCULATIONS FIRST --- block down to your sprintf(timeString...); statement, and replace it with this hybrid logic:
cpp
    // ==========================================================
    // --- HYBRID TIME CALCULATIONS (GPS + DS3231 RTC MOD) ---
    // ==========================================================
    int localHour = 12;
    int localMinute = 0;
    const float EASTERN_TIME_OFFSET = -4.0; // Adjust for your local timezone

    // STRATEGY 1: GPS Has Lock -> Read satellite time and update the RTC chip
    if (gps.time.isValid() && gps.satellites.value() > 0) {
        int rawHour = gps.time.hour() + (int)EASTERN_TIME_OFFSET;
        localMinute = gps.time.minute() + (int)((EASTERN_TIME_OFFSET - (int)EASTERN_TIME_OFFSET) * 60);

        if (localMinute >= 60) { localMinute -= 60; rawHour += 1; }
        else if (localMinute < 0) { localMinute += 60; rawHour -= 1; }
        if (rawHour >= 24) { rawHour -= 24; }
        else if (rawHour < 0) { rawHour += 24; }
        
        localHour = rawHour;

        // Auto-sync the RTC chip memory with the fresh satellite timestamp once per loop
        if (rtcFound && gps.date.isValid()) {
            rtc.adjust(DateTime(gps.date.year(), gps.date.month(), gps.date.day(), gps.time.hour(), gps.time.minute(), gps.time.second()));
        }
    } 
    // STRATEGY 2: No GPS Lock -> Pull time instantly from the hardware RTC module
    else if (rtcFound) {
        DateTime now = rtc.now();
        
        // Apply timezone calculation to the raw UTC time stored on the RTC
        int rawHour = now.hour() + (int)EASTERN_TIME_OFFSET;
        localMinute = now.minute();
        
        if (rawHour >= 24) { rawHour -= 24; }
        else if (rawHour < 0) { rawHour += 24; }
        
        localHour = rawHour;
    } 
    // STRATEGY 3: Fail-safe backup fallback if everything is disconnected
    else {
        unsigned long totalSecs = currentMillis / 1000;
        localMinute = (totalSecs / 60) % 60;
        int rawHour = (12 + (totalSecs / 3600)) % 24;
        localHour = (rawHour + (int)EASTERN_TIME_OFFSET + 24) % 24;
    }

    // Convert to 12-Hour format for presentation layout
    int displayHour = localHour % 12;
    if (displayHour == 0) displayHour = 12;

    // Format time text string for the EEZ engine
    char timeString; 
    sprintf(timeString, "%d:%02d", displayHour, localMinute); 
    eez::flow::setGlobalVariable(FLOW_GLOBAL_VARIABLE_TIME_DATA, eez::StringValue(timeString));
    // ==========================================================


Physical Wiring Reminder

When your mini DS3231 board arrives, wire its pins to your Heltec board as follows:
• VCC → 3V3
• GND → GND
• SDA → B2B_SDA (GPIO pin connected to your BME280 data line)
• SCL → B2B_SCL (GPIO pin connected to your BME280 clock line)
